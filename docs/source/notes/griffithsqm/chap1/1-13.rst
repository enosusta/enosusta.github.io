    import numpy as np
    import matplotlib.pyplot as plt

    A = 1.0
    omega = 1.0
    number_of_samples = 10_000

    rng = np.random.default_rng(seed=20260826)
    times = rng.uniform(0.0, 2.0 * np.pi / omega, number_of_samples)
    positions = A * np.cos(omega * times)

    x = np.linspace(-0.995 * A, 0.995 * A, 1001)
    rho = 1.0 / (np.pi * np.sqrt(A**2 - x**2))

    plt.hist(
        positions,
        bins=100,
        range=(-A, A),
        density=True,
        alpha=0.45,
        label="10,000 samples",
    )
    plt.plot(x, rho, linewidth=2.0, label=r"$\rho(x)$")
    plt.xlabel(r"$x$")
    plt.ylabel("Probability density")
    plt.xticks([-A, 0.0, A], [r"$-A$", r"$0$", r"$A$"])
    plt.xlim(-1.05 * A, 1.05 * A)
    plt.grid(alpha=0.25)
    plt.legend()
    plt.show()
