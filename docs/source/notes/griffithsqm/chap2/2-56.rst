    import numpy as np
    import matplotlib.pyplot as plt
    from scipy.integrate import solve_ivp
    from scipy.optimize import brentq

    xi_max = 6.0

    def schrodinger_equation(xi, state, K):
        psi, derivative = state
        return derivative, (xi**2 - K) * psi

    def solve_wavefunction(K, initial_condition, xi_values=None):
        return solve_ivp(
            lambda xi, state: schrodinger_equation(xi, state, K),
            (0.0, xi_max),
            initial_condition,
            t_eval=xi_values,
            method="DOP853",
            rtol=1.0e-10,
            atol=1.0e-12,
        )

    def tail(K, initial_condition):
        solution = solve_wavefunction(K, initial_condition)
        return solution.y[0, -1]

    states = [
        (1, (0.0, 1.0), (2.9, 3.1)),
        (2, (1.0, 0.0), (4.9, 5.1)),
        (3, (0.0, 1.0), (6.9, 7.1)),
    ]

    energy_levels = []
    for n, initial_condition, bracket in states:
        K = brentq(
            lambda value: tail(value, initial_condition),
            bracket[0],
            bracket[1],
            xtol=1.0e-12,
        )
        energy_levels.append((n, K, K / 2.0, initial_condition))
        print(f"n={n}: K = {K:.4f}, E = {K / 2.0:.4f} hbar omega")

    xi = np.linspace(0.0, xi_max, 1201)
    for n, K, energy, initial_condition in energy_levels:
        psi = solve_wavefunction(K, initial_condition, xi).y[0]
        plt.plot(xi, psi, label=rf"$n={n},\ K={K:.4f}$")

    plt.axhline(0.0, color="black", linewidth=0.8)
    plt.xlabel(r"$\xi$")
    plt.ylabel(r"$\psi_n(\xi)$")
    plt.xlim(0.0, xi_max)
    plt.ylim(-1.3, 1.1)
    plt.grid(alpha=0.25)
    plt.legend()
    plt.show()
